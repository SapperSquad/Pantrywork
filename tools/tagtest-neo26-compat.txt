# NeoForge 26.1.x compat boot: Croptopia + EpheroLib + Aquaculture 2 (the two
# food mods ATM11 ships). Proves the forward bridge, both role chains (incl.
# the 26.x-new largemouth bass), the generated dialect union, and release
# purity. Recipe-level foreign-ingredient crafts are covered by the Fabric
# 26.x boots (FD Refabricated provides the foreign pair there).
# Run via tools\rcon.ps1 -File tools\tagtest-neo26-compat.txt
forceload add 0 0
setblock 8 -60 8 minecraft:chest
# --- forward bridge: croptopia dialect -> canonical ---
item replace block 8 -60 8 container.0 with croptopia:cheese
execute if items block 8 -60 8 container.0 #c:foods/cheese
execute if items block 8 -60 8 container.0 #pantrywork:food_component/dairy
# --- aquaculture through canonical (bass is new in 2.9.x) ---
item replace block 8 -60 8 container.0 with aquaculture:largemouth_bass
execute if items block 8 -60 8 container.0 #c:foods/raw_fish
# --- generated dialect union loads on 26.x ---
execute if items block 8 -60 8 container.0 #c:fishes
# --- role chain: protein is cooked-only by design, use the cooked fillet ---
item replace block 8 -60 8 container.0 with aquaculture:fish_fillet_cooked
execute if items block 8 -60 8 container.0 #pantrywork:food_component/protein
# --- negative control ---
item replace block 8 -60 8 container.0 with minecraft:stone
execute unless items block 8 -60 8 container.0 #pantrywork:food_component
# --- release purity: dev-only recipe absent (crafter consumes nothing) ---
kill @e[type=item]
setblock 8 -60 12 minecraft:crafter[orientation=up_north]
item replace block 8 -60 12 container.0 with minecraft:bread
item replace block 8 -60 12 container.1 with minecraft:cooked_beef
item replace block 8 -60 12 container.2 with minecraft:carrot
setblock 9 -60 12 minecraft:redstone_block
time query gametime
time query gametime
execute if items block 8 -60 12 container.0 minecraft:bread
kill @e[type=item]
setblock 9 -60 12 minecraft:air
setblock 8 -60 12 minecraft:air
