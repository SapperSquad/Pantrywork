# Expected on the full boot: 119 passed / 0 failed / 0 expected.

# src/generated before the first boot (tools scratch build_suite.py), so a live failure

# analogue IS still in Pam's own c:rawbeef / c:butter / c:milk - that is upstream's own file

# ====================== PHASE F - Food Extended as a CONSUMER ======================
# The point of adding Food Extended: Pantrywork bridges none of its own items, but its 887
# recipes make 3472 tag reads (against only 178 exact-item reads) and 743 of those land in
# 31 tags Pantrywork writes. Both crafts below are its own shapeless recipes, made entirely
# from OTHER mods' food reaching its ingredient tags.
#
# Craft 1: friedonionsitem = c:tool_pot + c:crops/onion + c:butter,
# built from CROPTOPIA's onion and CULTURAL DELIGHTS' butter.
kill @e[type=item]
setblock 8 -60 12 minecraft:crafter[orientation=up_north]
item replace block 8 -60 12 container.0 with pamhc2foodcore:potitem
item replace block 8 -60 12 container.1 with croptopia:onion
item replace block 8 -60 12 container.2 with culturaldelights:butter
setblock 9 -60 12 minecraft:redstone_block
# tick pad: with Create/BF/FoT/Refurbished loaded the crafter tick lands after the next
# RCON command, so an unpadded contents check races it (see CLAUDE.md).
time query gametime
time query gametime
execute if items entity @e[type=minecraft:item,limit=1] contents pamhc2foodextended:friedonionsitem
kill @e[type=item]
setblock 9 -60 12 minecraft:air
setblock 8 -60 12 minecraft:air

setblock 9 -60 12 minecraft:air
